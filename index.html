🛠️ 3. SQL-Tabelle in Supabase anlegen

Führe diesen Befehl 1× im Supabase SQL Editor aus (legt Tabelle m2_progress und Row Level Security an): 

create table if not exists m2_progress (
  user_id uuid primary key references auth.users(id) on delete cascade,
  day_status jsonb default '{}'::jsonb,
  day_comments jsonb default '{}'::jsonb,
  quiz_scores jsonb default '{}'::jsonb,
  repeat_flags jsonb default '{}'::jsonb,
  calendar_state jsonb default '{}'::jsonb,
  updated_at timestamptz default timezone('utc'::text, now()) not null
);

alter table m2_progress enable row level security;

create policy "Nutzer verwaltet nur eigene M2-Daten"
  on m2_progress for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
